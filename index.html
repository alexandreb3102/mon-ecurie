/* ===================================================== NAISSANCE D'UN POULAIN - Prénom choisi par le joueur - Même race x même race = même race - Deux races différentes = OC - Parents enregistrés dans le carnet - Carnet accessible immédiatement ===================================================== */ function createFoal( motherId, fatherId ){ let m= game.horses.find( h=>h.id==motherId ); let f= game.horses.find( h=>h.id==fatherId ); if(!m||!f) return; /* Vérification du budget */ if(game.money<1000){ openModal( "💸 Reproduction impossible", `
Il faut 1 000 € pour lancer la reproduction.
` ); return; } /* ===================================================== CALCUL DU POTENTIEL ===================================================== */ let geneticAverage= ( m.potential+ f.potential )/2; let geneticVariation= Math.random()*12-6; let pot= clamp( Math.round( geneticAverage+ geneticVariation ), 40, 99 ); /* ===================================================== DÉTERMINATION DE LA RACE ===================================================== Même race : Selle Français + Selle Français → Selle Français Races différentes : Selle Français + Anglo-Arabe → OC ===================================================== */ let foalRace; if(m.race===f.race){ foalRace=m.race; }else{ foalRace="OC"; } /* ===================================================== SEXE ===================================================== */ let foalSex= Math.random()<0.5 ?"Jument" :"Étalon"; /* ===================================================== CRÉATION DU POULAIN ===================================================== */ let foal= makeHorse( "h"+Date.now(), "", foalRace, 0, foalSex, pot, 20, 0, "pasture" ); /* ===================================================== POTENTIEL RÉEL ===================================================== */ foal.realPotential=pot; foal.potential=pot; foal.currentLevel=20; foal.experience=0; /* ===================================================== CARACTÉRISTIQUES HÉRITÉES ===================================================== */ foal.technique= clamp( Math.round( (m.technique+f.technique)/2+ (Math.random()*10-5) ), 40, 99 ); foal.strength= clamp( Math.round( (m.strength+f.strength)/2+ (Math.random()*10-5) ), 40, 99 ); foal.endurance= clamp( Math.round( (m.endurance+f.endurance)/2+ (Math.random()*10-5) ), 40, 99 ); foal.regularity= clamp( Math.round( (m.regularity+f.regularity)/2+ (Math.random()*10-5) ), 40, 99 ); /* ===================================================== MENTAL HÉRITÉ ===================================================== */ foal.mental= clamp( Math.round( (m.mental+f.mental)/2+ (Math.random()*10-5) ), 40, 99 ); foal.gentleness= clamp( Math.round( (m.gentleness+f.gentleness)/2+ (Math.random()*10-5) ), 40, 99 ); foal.calm= clamp( Math.round( (m.calm+f.calm)/2+ (Math.random()*10-5) ), 40, 99 ); foal.courage= clamp( Math.round( (m.courage+f.courage)/2+ (Math.random()*10-5) ), 40, 99 ); foal.concentration= clamp( Math.round( (m.concentration+f.concentration)/2+ (Math.random()*10-5) ), 40, 99 ); foal.willpower= clamp( Math.round( (m.willpower+f.willpower)/2+ (Math.random()*10-5) ), 40, 99 ); foal.reactivity= clamp( Math.round( (m.reactivity+f.reactivity)/2+ (Math.random()*10-5) ), 20, 99 ); /* ===================================================== UTILITÉ ===================================================== */ foal.courseUse= clamp( Math.round( (m.courseUse+f.courseUse)/2+ (Math.random()*10-5) ), 40, 99 ); foal.competitionUse= clamp( Math.round( (m.competitionUse+f.competitionUse)/2+ (Math.random()*10-5) ), 40, 99 ); foal.breedingUse= clamp( Math.round( (m.breedingUse+f.breedingUse)/2+ (Math.random()*10-5) ), 40, 99 ); /* ===================================================== SANTÉ DU POULAIN ===================================================== */ foal.form=90; foal.health= clamp( Math.round( (m.health+f.health)/2+ (Math.random()*6-3) ), 70, 100 ); foal.fatigue=0; /* ===================================================== SUIVI PROFESSIONNEL ===================================================== */ foal.hoof={ type:"parage", lastDay:game.day, nextDay:game.day+60 }; foal.vet={ lastDay:game.day, nextDay:game.day+30 }; foal.dentist={ lastDay:game.day, nextDay:game.day+180 }; foal.osteo={ lastDay:game.day, nextDay:game.day+120 }; /* ===================================================== ORIGINES ===================================================== */ foal.father={ name:f.name, race:f.race, results:[ `Race : ${f.race}`, `Potentiel : ${f.potential}/100`, `Niveau actuel : ${f.currentLevel}/100`, `${wins(f)} victoire(s)` ] }; foal.mother={ name:m.name, race:m.race, results:[ `Race : ${m.race}`, `Potentiel : ${m.potential}/100`, `Niveau actuel : ${m.currentLevel}/100`, `${wins(m)} victoire(s)` ] }; /* ===================================================== IMPORTANT : ON DEMANDE LE PRÉNOM AVANT DE DÉBITER L'ARGENT ===================================================== */ openModal( "🐣 Naissance !", `
🐴
🎉 Un poulain vient de naître !
Sexe : ${foal.sex}

Race : ${foal.race} ${ foal.race==="OC" ? `
ℹ️ Les parents sont de deux races différentes : le poulain est automatiquement OC (Origines Constatées). ` : `
ℹ️ Les deux parents sont de la même race : le poulain conserve cette race. ` }
🧬 Père
${f.name} · ${f.race}

🧬 Mère
${m.name} · ${m.race}
✍️ Choisis le prénom du poulain
Le prénom choisi sera enregistré définitivement dans son carnet.
` ); } /* ===================================================== VALIDATION DU PRÉNOM ===================================================== */ function validateFoalName( foalData, motherId, fatherId ){ let input= document.getElementById( "foalNameInput" ); if(!input) return; let name= input.value.trim(); /* Prénom obligatoire */ if(!name){ openModal( "✍️ Prénom obligatoire", `
Tu dois choisir un prénom pour le poulain avant de continuer.
` ); return; } /* Longueur */ if(name.length<2){ openModal( "✍️ Prénom trop court", `
Le prénom doit contenir au moins 2 caractères.
` ); return; } /* Vérification des doublons */ let alreadyExists= game.horses.some( h=> h.name.toLowerCase()=== name.toLowerCase() ); if(alreadyExists){ openModal( "✍️ Prénom déjà utilisé", `
Un cheval porte déjà le prénom ${name}. Choisis un autre prénom.
` ); return; } /* ===================================================== CRÉATION DÉFINITIVE ===================================================== */ let foal= clone(foalData); foal.name=name; /* Paiement */ game.money-=1000; /* Ajout à l'écurie */ game.horses.push(foal); /* Sauvegarde */ save(); /* ===================================================== OUVERTURE DU CARNET ===================================================== */ openModal( "🐣 "+foal.name+" est né !", `
🐴
🎉 Félicitations !

${foal.name} est maintenant enregistré dans ton écurie.
🐴 ${foal.name}
${foal.sex} · ${foal.race} · 0 an

🔎 Potentiel estimé : ${potentialRange(foal)[0]}–${potentialRange(foal)[1]}
🧬 Père : ${foal.father.name}
${foal.father.race}

🧬 Mère : ${foal.mother.name}
${foal.mother.race}
` ); }
